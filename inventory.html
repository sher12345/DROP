Supabase SQL Setup

Run this once in your Supabase → SQL Editor. It reuses the same products / orders / order_items tables as the Drop POS and only adds the inventory-specific tables plus a few columns on products.

-- ═══════════════════════════════════════════
-- DROP INVENTORY — SUPABASE SQL SETUP
-- Run once. Shares the products/orders/order_items
-- tables already created for the Drop POS.
-- ═══════════════════════════════════════════

-- Extra columns on the existing products table
alter table products add column if not exists cost_price numeric default 0;
alter table products add column if not exists supplier text default '';
alter table products add column if not exists reorder_point numeric default 5;

-- 1. RESTOCK (GRN) HEADER
create table if not exists inv_grn (
  id           text primary key,
  bizday       date not null,
  supplier     text not null,
  invoiceno    text default '',
  grandtotal   numeric default 0,
  notes        text default '',
  createdat    timestamptz default now()
);

-- 2. RESTOCK (GRN) ITEMS
create table if not exists inv_grn_items (
  id            text primary key,
  grn_id        text references inv_grn(id) on delete cascade,
  product_id    uuid,
  sku           text not null,
  itemname      text not null,
  size          text not null,
  qty           numeric not null,
  unitcost      numeric not null,
  linetotal     numeric not null
);

-- 3. STOCK TAKE HEADER
create table if not exists inv_stocktake (
  id              text primary key,
  bizday          date not null,
  countedby       text not null,
  totalvariancevalue numeric default 0,
  createdat       timestamptz default now()
);

-- 4. STOCK TAKE ITEMS
create table if not exists inv_stocktake_items (
  id             text primary key,
  stocktake_id   text references inv_stocktake(id) on delete cascade,
  product_id     uuid,
  sku            text not null,
  itemname       text not null,
  size           text not null,
  system_stock   numeric default 0,
  physical       numeric default 0,
  variance       numeric default 0,
  costperunit    numeric default 0,
  variancevalue  numeric default 0
);

-- 5. EXPENSES (till cash)
create table if not exists inv_expenses (
  id          text primary key,
  bizday      date not null,
  category    text not null default 'general',
  description text not null default '',
  amount      numeric not null default 0,
  paidby      text default '',
  notes       text default '',
  createdat   timestamptz default now()
);

-- 6. INVENTORY APP USERS
create table if not exists inv_users (
  id          text primary key,
  username    text unique not null,
  password    text not null,
  role        text not null default 'staff',
  active      boolean default true,
  permissions jsonb default '{}',
  createdat   timestamptz default now()
);

-- Enable Realtime (optional)
alter publication supabase_realtime add table inv_grn;
alter publication supabase_realtime add table inv_stocktake;
alter publication supabase_realtime add table inv_expenses;
